[1m[33m12.5 Bank Account Class[0m
[90m-----------------------[0m
A small class with methods and a class counter:

[32m    class Account[0m
[32m      @@total = 0[0m

[32m      def initialize(owner, balance = 0)[0m
[32m        @owner = owner[0m
[32m        @balance = balance[0m
[32m        @@total += 1[0m
[32m      end[0m

[32m      def deposit(amount)[0m
[32m        @balance += amount[0m
[32m      end[0m

[32m      def withdraw(amount)[0m
[32m        @balance -= amount if @balance >= amount[0m
[32m      end[0m

[32m      attr_reader :balance[0m

[32m      def self.count[0m
[32m        @@total[0m
[32m      end[0m
[32m    end[0m

[32m    a = Account.new("Ada", 100)[0m
[32m    a.deposit(50)[0m
[32m    a.withdraw(30)[0m
[32m    puts a.balance      # => 120[0m
[32m    puts Account.count  # => 1[0m

TRY IT: create a second account and confirm that Account.count goes
